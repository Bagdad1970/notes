class TrieNode:  
    def __init__(self):  
        self.children = {}  
        self.end_of_word = False  
        self.value = 0  
  
    def get_children(self):  
        return self.children  
  
    def set_end_of_word(self, end_of_word):  
        self.end_of_word = end_of_word  
  
    def get_value(self):  
        return self.value  
  
    def set_value(self, value):  
        self.value = value  
  
class Trie:  
    def __init__(self):  
        self.root = TrieNode()  
        self.counter = 0  
  
    def insert(self, key: str, value: int) -> None:  
        current = self.root  
  
        for char in key:  
            if char not in current.get_children():  
                current.get_children()[char] = TrieNode()  
            current = current.get_children()[char]  
  
        current.set_end_of_word(True)  
        current.set_value(value)  
  
    def find_prefix(self, prefix: str, index: int) -> int:  
        current = self.root  
        self.counter = 0  
  
        for i in range(len(prefix)):  
            ch = prefix[i]  
            node = current.get_children().get(ch)  
            if node is None:  
                return -1  
            current = node  
  
        return self._find_rec(current, index) + 1  
  
    def _find_rec(self, root: TrieNode, index: int) -> int:  
        if root.end_of_word:  
            self.counter += 1  
  
        if self.counter == index:  
            return root.get_value()  
  
        result = -1  
        for child in root.children.values():  
            result = self._find_rec(child, index)  
            if result > 0:  
                return result  
  
        return result  
  
  
n, q = map(int, input().split())  
  
tree = Trie()  
for i in range(n):  
    tree.insert(input(), i)  
  
queries = []  
for i in range(q):  
    query = input().split()  
    k, prefix = int(query[0]), query[1]  
    queries.append( (k, prefix) )  
  
  
for query in queries:  
    k, prefix = query  
      
    print(tree.find_prefix(prefix, k))
