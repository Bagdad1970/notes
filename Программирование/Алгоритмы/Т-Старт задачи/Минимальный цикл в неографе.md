from collections import deque  
  
def shortest_simple_cycle(n, graph):  
    INF = float('inf')  
    answer = INF  
  
    for start in range(n):  
        dist = [-1] * n  
        parent = [-1] * n  
  
        queue = deque([start])  
        dist[start] = 0  
  
        while queue:  
            u = queue.popleft()  
  
            for v in graph[u]:  
                if dist[v] == -1:  
                    dist[v] = dist[u] + 1  
                    parent[v] = u  
                    queue.append(v)  
                elif parent[u] != v:  
                    cycle_len = dist[u] + dist[v] + 1  
                    if cycle_len < answer:  
                        answer = cycle_len  
  
    return answer if answer != INF else -1  
  
  
n, m = map(int, input().split())  
graph = [[] for _ in range(n)]  
  
for _ in range(m):  
    a, b = map(int, input().split())  
    a -= 1  
    b -= 1  
    graph[a].append(b)  
    graph[b].append(a)  
  
print(shortest_simple_cycle(n, graph))
